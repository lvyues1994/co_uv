cmake_minimum_required(VERSION 3.14)
project(uvco_consumer LANGUAGES C CXX)
if(UVCO_CONSUMER_MODE STREQUAL "subdirectory")
    add_subdirectory("${UVCO_SOURCE_DIR}" uvco-source)
elseif(UVCO_CONSUMER_MODE STREQUAL "fetchcontent")
    include(FetchContent)
    FetchContent_Declare(uvco SOURCE_DIR "${UVCO_SOURCE_DIR}")
    FetchContent_MakeAvailable(uvco)
else()
    find_package(uvco 0.1 CONFIG REQUIRED)
endif()
if(UVCO_SOURCE_DIR AND (UVCO_BUILD_TESTS OR UVCO_ENABLE_INSTALL))
    message(FATAL_ERROR "uvco subproject must default tests/install OFF")
endif()
add_executable(consumer main.cpp other.cpp)
target_link_libraries(consumer PRIVATE uvco::uvco)
enable_testing()
add_test(NAME consumer-two-tu COMMAND consumer)
