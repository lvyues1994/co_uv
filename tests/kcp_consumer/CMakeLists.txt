cmake_minimum_required(VERSION 3.14)
project(kcp_consumer LANGUAGES C CXX)
if(UVCO_SOURCE_DIR)
    set(UVCO_ENABLE_KCP ON CACHE BOOL "" FORCE)
    if(UVCO_CONSUMER_MODE STREQUAL "fetchcontent")
        include(FetchContent)
        FetchContent_Declare(uvco SOURCE_DIR "${UVCO_SOURCE_DIR}")
        FetchContent_MakeAvailable(uvco)
    else()
        add_subdirectory("${UVCO_SOURCE_DIR}" uvco-source)
    endif()
else()
    find_package(uvco 0.1 CONFIG REQUIRED COMPONENTS kcp)
endif()
if(NOT TARGET uvco::kcp)
    message(FATAL_ERROR "KCP component did not provide uvco::kcp")
endif()
add_executable(kcp-consumer main.cpp other.cpp)
target_link_libraries(kcp-consumer PRIVATE uvco::kcp)
enable_testing()
add_test(NAME kcp-consumer-two-tu COMMAND kcp-consumer)
